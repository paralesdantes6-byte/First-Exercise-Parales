Binary Search

namespace Binary_search
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void lblTitle_Click(object sender, EventArgs e)
        {

        }

        private void Form1_Load(object sender, EventArgs e)
        {

        }

        private void btnOlogn_Click(object sender, EventArgs e)
        {
            int[] numbers = { 10, 20, 30, 40, 50, 60, 70 };
            int target = 50;
            int left = 0;
            int right = numbers.Length - 1;
            bool found = false;

            while (left <= right)
            {
                int middle = (left + right) / 2;

                if (numbers[middle] == target)
                {
                    found = true;
                    break;
                }
                else if (numbers[middle] < target)
                {
                    left = middle + 1;
                }
                else
                {
                    right = middle - 1;
                }
            }

            if (found)
            {
                MessageBox.Show(
                    target + " was found!",
                    "Binary Search"
                );
            }
            else
            {
                MessageBox.Show(
                    "Number not found.",
                    "Binary Search"
                );
            }
        }



    }
}

            int low = 1;
            int high = 100;
            int guess = 50;

            string message = "Guessing a number from 1 to 100:\n\n";

            while (low <= high)
            {
                guess = (low + high) / 2;

                message += "My guess is " + guess + "\n";

                if (guess == 50)
                {
                    break;
                }

                high = guess - 1;
            }

            MessageBox.Show(
                message + "\nThe number is 50!",
                "Guessing Algorithm"
            );
        }
    }

namespace _15
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void label2_Click(object sender, EventArgs e)
        {

        }

        private void Form1_Load(object sender, EventArgs e)
        {

        }

        private void btnO2n_Click(object sender, EventArgs e)
        {
            string[] toppings = { "Cheese", "Mushroom", "Pepperoni" };
            string result = "PIZZA TOPPING COMBINATIONS\n\n";

            int total = 1 << toppings.Length;

            for (int i = 0; i < total; i++)
            {
                string pizza = "";

                for (int j = 0; j < toppings.Length; j++)
                {
                    if ((i & (1 << j)) != 0)
                    {
                        pizza += toppings[j] + ", ";
                    }
                }

                if (pizza == "")
                {
                    pizza = "Plain Pizza";
                }
                else
                {
                    pizza = pizza.TrimEnd(',', ' ');
                }

                result += (i + 1) + ". " + pizza + "\n";
            }

            result += "\nTotal combinations: " + total;

            MessageBox.Show(result, "Pizza Combinations");
        }
    }
}
